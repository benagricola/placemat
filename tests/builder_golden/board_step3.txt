"""Demo: layout."""
from placemat import board, Along, Beside, Edge, LinkWeight, OnEdge, PadRef, Part

# The board's width, chosen in the studio's board builder.
BOARD_WIDTH_MM = 60.0
# The board's height, chosen in the studio's board builder.
BOARD_HEIGHT_MM = 40.0
# The gap between U1 and J1: the connector's shell is tall. Chosen in the studio's board builder.
U1_GAP_MM = 0.5

board.rect(width=BOARD_WIDTH_MM, height=BOARD_HEIGHT_MM)

board.place(Part("j1"), at=OnEdge(Edge.NORTH, along=Along.MID))
board.place(Part("u1"), at=Beside(Part("j1"), Edge.SOUTH, gap=U1_GAP_MM))

# Searched from their links.
board.link(PadRef(Part("c1"), "VIN"), PadRef(Part("u1"), "VIN"), weight=LinkWeight.SHORT)
board.place(Part("c1"))
